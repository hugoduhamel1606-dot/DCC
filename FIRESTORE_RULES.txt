rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    match /config/{id} {
      allow read, write: if request.auth != null;
    }

    match /users/{userId} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
    }

    match /projects/{projectId} {
      // Lecture : uid dans sharedWith OU email dans sharedEmails
      allow read: if request.auth != null && (
        request.auth.uid in resource.data.sharedWith ||
        request.auth.token.email in resource.data.sharedEmails
      );
      // Écriture : uid dans sharedWith (créateur ou déjà partagé)
      allow write: if request.auth != null && (
        !resource.data.keys().hasAll(['sharedWith']) ||
        request.auth.uid in resource.data.sharedWith ||
        request.auth.token.email in resource.data.sharedEmails
      );
      // Création
      allow create: if request.auth != null;

      match /activity/{id}  { allow read, write: if request.auth != null; }
      match /versions/{id}  { allow read, write: if request.auth != null; }
      match /sampleSets/{ssId} {
        allow read, write: if request.auth != null;
        match /spectra/{specId} { allow read, write: if request.auth != null; }
      }
      match /models/{id} { allow read, write: if request.auth != null; }
    }
  }
}
